function doPost(e) { // ★★★ 아래 따옴표 안에 복사한 구글 스프레드시트의 주소를 붙여넣으세요! ★★★ var sheetUrl = "https://docs.google.com/spreadsheets/d/1epODoCoQTNBOAoCIbMlJx-yyHn3l4mH0OnjOalmuUT0/edit?gid=0#gid=0"; var ss = SpreadsheetApp.openByUrl(sheetUrl); var sheet = ss.getActiveSheet(); var data = JSON.parse(e.postData.contents); // 1. Base64 텍스트를 이미지 파일(Blob)로 변환 var base64Data = data.signature.split(",")[1]; var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), 'image/png', data.name + '_서명.png'); // 2. 구글 드라이브 최상단(내 드라이브)에 이미지 파일 저장 var folder = DriveApp.getRootFolder(); var file = folder.createFile(blob); // 3. 시트 셀에서 이미지를 불러올 수 있도록 링크 보기 권한 허용 file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW); // 4. 셀 안에 쏙 들어가게 만드는 IMAGE 함수 생성 var fileId = file.getId(); var imageUrl = "https://drive.google.com/uc?export=view&id=" + fileId; var imageFormula = '=IMAGE("' + imageUrl + '")'; // 5. 시트의 마지막 줄에 데이터 추가 sheet.appendRow([ new Date(), data.date, data.affiliation, data.name, imageFormula ]); // 6. 이미지가 잘 보이도록 방금 추가된 행의 높이를 늘려줌 (60픽셀) sheet.setRowHeight(sheet.getLastRow(), 60); return ContentService.createTextOutput(JSON.stringify({"result": "success"})) .setMimeType(ContentService.MimeType.JSON); }
